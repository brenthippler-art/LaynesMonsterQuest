+import { PixelBuffer, rgba, stamp } from '../../render/pixels';
+import { realmIconColor } from '../../ui/pixelIcon';
+import { isWalkable, type MapRide, type TileMap } from '../../world/tilemap';
+import { type BuildingPlan, type RidePlan } from './types';
+
+/**
+ * Turn a plan's rides into MapRides the renderer draws every frame.
+ *
+ * Each sprite is turned into a tiny picture only once (the first time it's drawn),
+ * so a frame is just a few drawImage() calls per ride on screen: cheap, even on an
+ * older tablet. Positions are rounded to whole art pixels, so rides move one
+ * pixel-art pixel at a time, like everything else in the game.
+ */
+export function attachRides(plan: BuildingPlan, maps: TileMap[]): string[] {
+  const problems: string[] = [];
+  for (const ride of plan.rides ?? []) {
+    const map = maps.find((m) => m.id === `${plan.id}/${ride.map}`);
+    if (!map) throw new Error(`World "${plan.id}": a ride is on map "${ride.map}", which isn't in the maps list.`);
+    for (const part of ride.pose(0)) {
+      if (!ride.sprites[part.sprite]) throw new Error(`World "${plan.id}": a ride uses sprite "${part.sprite}", which it doesn't have.`);
+    }
+    problems.push(...walkableUnder(ride, map));
+    (map.rides ??= []).push(toMapRide(ride));
+  }
+  return problems;
+}
+
+function toMapRide(ride: RidePlan): MapRide {
+  const pictures = new Map<string, HTMLCanvasElement>();
+  const picture = (name: string) => {
+    let canvas = pictures.get(name);
+    if (!canvas) {
+      const rows = ride.sprites[name];
+      const buf = new PixelBuffer(Math.max(...rows.map((r) => r.length)), rows.length);
+      stamp(buf, rows, (ch) => rgba(realmIconColor(ch)), 0, 0);
+      canvas = buf.toCanvas();
+      pictures.set(name, canvas);
+    }
+    return canvas;
+  };
+  return {
+    x: ride.x,
+    y: ride.y,
+    w: ride.w,
+    h: ride.h,
+    draw(ctx, sx, sy, p, time) {
+      for (const part of ride.pose(time)) {
+        const pic = picture(part.sprite);
+        ctx.drawImage(pic, sx + Math.round(part.x) * p, sy + Math.round(part.y) * p, pic.width * p, pic.height * p);
+      }
+    },
+  };
+}

+/** A ride drawn over a walkable square would hide Layne (and could look like it's in the way). */
+function walkableUnder(ride: RidePlan, map: TileMap): string[] {
+  for (let y = ride.y; y < ride.y + ride.h; y++) {
+    for (let x = ride.x; x < ride.x + ride.w; x++) {
+      if (isWalkable(map, x, y)) return [`Map "${map.id}": the ride at ${ride.x},${ride.y} covers ${x},${y}, a square Layne can walk on.`];
+    }
+  }
+  return [];
+}

+import { rect } from '../building/paint';
+import { type TileCtx } from '../building/types';
+
+/**
+ * Wonderwhirl Park's own sign: a swirly pinwheel, drawn in code (so it's nobody
+ * else's logo). Its blades curl as they go out, like a lollipop swirl.
+ *
+ * pinwheelPick() says which blade a pixel is on: dx, dy = how far from the middle,
+ * r = how big, turn = how far it has spun (in radians). -1 = outside the pinwheel.
+ */
+const BLADES = 6;
+const CURL = 2.2; // how much the blades curl toward the edge
+
+export function pinwheelPick(dx: number, dy: number, r: number, turn: number): number {
+  const d = Math.hypot(dx, dy);
+  if (d > r) return -1;
+  if (d < r * 0.18) return BLADES; // the round middle
+  const a = Math.atan2(dy, dx) - turn + (CURL * d) / r;
+  const slice = (((a / (Math.PI * 2)) * BLADES) % BLADES + BLADES) % BLADES;
+  return Math.floor(slice);
+}
+
+/**
+ * The pinwheel as rows of decoration letters (for the spinning sign and the world card).
+ * `letters` = the blades' colors, going round; `middle` = the dot in the center.
+ */
+export function pinwheelRows(size: number, turn: number, letters: string, middle: string, rim = 'k'): string[] {
+  const r = size / 2 - 0.5;
+  const rows: string[] = [];
+  for (let y = 0; y < size; y++) {
+    let row = '';
+    for (let x = 0; x < size; x++) {
+      const dx = x - r;
+      const dy = y - r;
+      const pick = pinwheelPick(dx, dy, r, turn);
+      if (pick < 0) row += '.';
+      else if (Math.hypot(dx, dy) > r - 1) row += rim; // a dark outline round the edge
+      else row += pick === BLADES ? middle : letters[pick % letters.length];
+    }
+    rows.push(row);
+  }
+  return rows;
+}
+
+/** A small pinwheel painted straight into a tile (the swirls on the sign board). */
+export function drawPinwheel(c: TileCtx, cx: number, cy: number, r: number, turn: number, colors: string[]): void {
+  for (let y = -r; y <= r; y++) {
+    for (let x = -r; x <= r; x++) {
+      const pick = pinwheelPick(x, y, r, turn);
+      if (pick >= 0 && pick < BLADES && pick % 2 === 0) rect(c, cx + x, cy + y, 1, 1, colors[(pick / 2) % colors.length]);
+    }
+  }
+}

+/** A point that goes round a rectangle (corners at x0,y0 and x1,y1) at `d` pixels from the top-left, clockwise. */
+function aroundBox(d: number, x0: number, y0: number, x1: number, y1: number): { x: number; y: number; across: boolean } {
+  const w = x1 - x0;
+  const h = y1 - y0;
+  const p = wrap(d, 2 * (w + h));
+  if (p < w) return { x: x0 + p, y: y0, across: true };
+  if (p < w + h) return { x: x1, y: y0 + p - w, across: false };
+  if (p < 2 * w + h) return { x: x1 - (p - w - h), y: y1, across: true };
+  return { x: x0, y: y1 - (p - 2 * w - h), across: false };
+}

+/**
+ * A tiny drawing pad for making ride pictures in code: a grid of decoration
+ * letters ('.' = see-through) with a few pen tools. rows() gives the finished
+ * picture, ready to use as a ride sprite.
+ */
+export class Sketch {
+  private grid: string[][];
+
+  constructor(
+    readonly w: number,
+    readonly h: number,
+  ) {
+    this.grid = Array.from({ length: h }, () => Array<string>(w).fill('.'));
+  }
+
+  dot(x: number, y: number, ch: string): void {
+    const px = Math.round(x);
+    const py = Math.round(y);
+    if (px >= 0 && py >= 0 && px < this.w && py < this.h) this.grid[py][px] = ch;
+  }
+
+  rect(x: number, y: number, w: number, h: number, ch: string): void {
+    for (let py = y; py < y + h; py++) for (let px = x; px < x + w; px++) this.dot(px, py, ch);
+  }
+
+  /** A straight line, `thick` pixels wide. */
+  line(x0: number, y0: number, x1: number, y1: number, ch: string, thick = 1): void {
+    const steps = Math.max(1, Math.ceil(Math.hypot(x1 - x0, y1 - y0)));
+    for (let i = 0; i <= steps; i++) {
+      const x = x0 + ((x1 - x0) * i) / steps;
+      const y = y0 + ((y1 - y0) * i) / steps;
+      this.rect(Math.round(x - (thick - 1) / 2), Math.round(y - (thick - 1) / 2), thick, thick, ch);
+    }
+  }
+
+  /** A ring: every pixel between radius r - width and r from (cx, cy). */
+  ring(cx: number, cy: number, r: number, width: number, ch: string): void {
+    for (let y = Math.floor(cy - r); y <= Math.ceil(cy + r); y++) {
+      for (let x = Math.floor(cx - r); x <= Math.ceil(cx + r); x++) {
+        const d = Math.hypot(x - cx, y - cy);
+        if (d <= r && d > r - width) this.dot(x, y, ch);
+      }
+    }
+  }
+
+  /** A filled circle. */
+  disc(cx: number, cy: number, r: number, ch: string): void {
+    this.ring(cx, cy, r, r + 1, ch);
+  }
+
+  /** Copy a picture (rows of letters) onto the pad, skipping its see-through pixels. */
+  paste(rows: readonly string[], ox: number, oy: number): void {
+    rows.forEach((row, y) => {
+      for (let x = 0; x < row.length; x++) if (row[x] !== '.') this.dot(ox + x, oy + y, row[x]);
+    });
+  }
+
+  rows(): string[] {
+    return this.grid.map((row) => row.join(''));
+  }
+}
+
+/** A picture mirrored left to right (a horse facing the other way). */
+export const mirror = (rows: readonly string[]): string[] => rows.map((row) => [...row].reverse().join(''));
+
+/** A picture turned a quarter turn (a coaster car going down instead of across). */
+export const turn = (rows: readonly string[]): string[] =>
+  Array.from({ length: rows[0].length }, (_, x) => rows.map((row) => row[x]).reverse().join(''));
+
+/** The same picture in other colors: swap = { from letter: to letter }. */
+export const recolor = (rows: readonly string[], swap: Record<string, string>): string[] =>
+  rows.map((row) => [...row].map((ch) => swap[ch] ?? ch).join(''));
+
+/** Wrap a number into 0..length (for things that go round and round). */
+export const wrap = (value: number, length: number): number => ((value % length) + length) % length;
